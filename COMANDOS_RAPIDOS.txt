━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 OPÇÃO 2: INSTALAÇÃO MANUAL PASSO A PASSO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# 1. Copiar plugin para container
sudo docker cp /home/diego/glpi11/flowBPMN 4a5500931c39:/var/www/glpi/plugins/

# 2. Ajustar permissões
sudo docker exec -it 4a5500931c39 bash -c "chown -R www-data:www-data /var/www/glpi/plugins/flowBPMN && chmod -R 755 /var/www/glpi/plugins/flowBPMN"

# 3. Instalar plugin
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:install flowbpmn"

# 4. Ativar plugin
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:activate flowbpmn"

# 5. Verificar instalação
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:list | grep flowbpmn"

 DESINSTALAR (SE NECESSÁRIO)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Desativar
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:deactivate flowbpmn"

# Desinstalar
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:uninstall flowbpmn"


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 LIMPAR E REINSTALAR DO ZERO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# 1. Desinstalar plugin atual (se instalado)
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:deactivate flowbpmn 2>/dev/null || true"
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:uninstall flowbpmn 2>/dev/null || true"

# 2. Remover diretório antigo
sudo docker exec -it 4a5500931c39 rm -rf /var/www/glpi/plugins/flowBPMN

# 3. Copiar versão atualizada
sudo docker cp /home/diego/glpi11/flowBPMN 4a5500931c39:/var/www/glpi/plugins/

# 4. Ajustar permissões
sudo docker exec -it 4a5500931c39 bash -c "chown -R www-data:www-data /var/www/glpi/plugins/flowBPMN && chmod -R 755 /var/www/glpi/plugins/flowBPMN"

# 5. Instalar novamente
sudo docker exec -it 4a5500931c39 bash -c "cd /var/www/glpi && php bin/console glpi:plugin:install flowbpmn && php bin/console glpi:plugin:activate flowbpmn"
